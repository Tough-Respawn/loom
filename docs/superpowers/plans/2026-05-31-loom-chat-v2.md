# Loom Chat v2 (Vision + Thinking) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ajouter à Loom Chat la lecture de screenshots (vision native Gemma 4 via mmproj) et l'affichage du raisonnement (thinking) de Gemma 4.

**Architecture:** On étend l'existant. `serve.py`/config activent le mmproj sur llama-server. `conversation.py` accepte un `content` multimodal. `client.py` streame des events taggés `reasoning`/`content`. `web/app.py` accepte une image et émet ces events en SSE. Le template affiche la réflexion (grisée, repliable) + la réponse, et permet de coller un screenshot.

**Tech Stack:** Python 3.12+, `uv`, `pytest`, `openai`, `flask`, llama.cpp (`--mmproj`), HTMX. Aucune nouvelle dépendance (base64 = stdlib).

**Spec de référence:** [docs/superpowers/specs/2026-05-31-loom-chat-v2-design.md](../specs/2026-05-31-loom-chat-v2-design.md)

> JAMAIS appeler un vrai serveur/modèle (tout est mocké). NE PAS télécharger le mmproj dans le
> workflow (c'est ~990 Mo, fait à la main au lancement réel).

```python
def _iter_events(stream) -> Iterator[tuple[str, str]]:
    """Yield ('reasoning', txt) ou ('content', txt) pour chaque delta non vide."""
    for chunk in stream:
        delta = chunk.choices[0].delta
        reasoning = getattr(delta, "reasoning_content", None)
        if reasoning:
            yield ("reasoning", reasoning)
        content = delta.content
        if content:
            yield ("content", content)
```

```python
    def stream_chat(
        self, messages: list[dict], system_prompt: str
    ) -> Iterator[tuple[str, str]]:

        full = [{"role": "system", "content": system_prompt}, *messages]
        stream = self._client.chat.completions.create(
            model=self.model, messages=full, stream=True
        )
        yield from _iter_events(stream)
```

Remplacer la fonction `chat()` par :
```python
    @app.post("/chat")
    def chat():
        message = (request.form.get("message") or "").strip()
        if not message or len(message) > 5000:
            return Response("message invalide", status=400)

        image = request.files.get("image")
        if image and image.filename:
            blob = image.read()
            if len(blob) > 10 * 1024 * 1024:
                return Response("image trop grande", status=400)
            mime = image.mimetype or "image/png"
            if not mime.startswith("image/"):
                return Response("fichier non-image", status=400)
            b64 = base64.b64encode(blob).decode("ascii")
            content = [
                {"type": "text", "text": message},
                {"type": "image_url",
                 "image_url": {"url": f"data:{mime};base64,{b64}"}},
            ]
        else:
            content = message

        conversation.add("user", content)
        conversation.save(history_path)

        def generate():
            answer = ""
            try:
                for kind, text in client.stream_chat(
                    conversation.to_messages(), conversation.system_prompt
                ):
                    if kind == "reasoning":
                        yield f"data: {json.dumps({'type': 'reasoning', 'text': text})}\n\n"
                    else:
                        answer += text
                        yield f"data: {json.dumps({'type': 'text', 'text': text})}\n\n"
                conversation.add("assistant", answer)
                conversation.save(history_path)
                yield f"data: {json.dumps({'type': 'done'})}\n\n"
            except Exception as exc:  # noqa: BLE001
                yield f"data: {json.dumps({'type': 'error', 'message': str(exc)})}\n\n"

        return Response(generate(), mimetype="text/event-stream")
```

```html
      const div = document.createElement('div');
      div.className = 'msg ' + role;
      div.innerHTML = html;
      messages.appendChild(div);
      window.scrollTo(0, document.body.scrollHeight);
      return div;
    }

    async function send(e) {
      e.preventDefault();
      const text = input.value.trim();
      if (!text) return false;

      let userHtml = text.replace(/</g, '&lt;');
      if (pendingImage) userHtml += '<img src="' + preview.src + '">';
      bubble('user', userHtml);

      const fd = new FormData();
      fd.append('message', text);
      if (pendingImage) fd.append('image', pendingImage);
      input.value = '';
      clearImg();

      const think = document.createElement('details');
      think.className = 'think';
      think.innerHTML = '<summary>🧠 réflexion…</summary><div></div>';
      messages.appendChild(think);
      const thinkBody = think.querySelector('div');
      const assistant = bubble('assistant', '');

      const resp = await fetch('/chat', { method: 'POST', body: fd });
      const reader = resp.body.getReader();
      const decoder = new TextDecoder();
      let buffer = '';
      while (true) {
        const { done, value } = await reader.read();
        if (done) break;
        buffer += decoder.decode(value, { stream: true });
        const parts = buffer.split('\n\n');
        buffer = parts.pop();
        for (const part of parts) {
          if (!part.startsWith('data: ')) continue;
          const evt = JSON.parse(part.slice(6));
          if (evt.type === 'reasoning') {
            thinkBody.textContent += evt.text;
          } else if (evt.type === 'text') {
            assistant.textContent += evt.text;
          } else if (evt.type === 'error') {
            assistant.className = 'msg assistant err';
            assistant.textContent = 'Erreur : ' + evt.message + ' — Loom est-il lancé ?';
          }
          window.scrollTo(0, document.body.scrollHeight);
        }
      }
      if (!thinkBody.textContent) think.remove();  // pas de réflexion -> on enlève le bloc
      return false;
    }
  </script>
</body>
</html>
```
