# Loom Chat (v1) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Une interface web Flask locale pour parler à Loom, avec mémoire de conversation persistée (JSON), reset, et vrai streaming token-par-token via le SDK OpenAI.

**Architecture:** Trois unités à responsabilité unique : `conversation.py` (mémoire : historique + persistance JSON), `client.py` (I/O modèle via le SDK `openai` pointé sur l'endpoint OpenAI-compatible de Loom), et `web/app.py` (Flask : page HTMX + endpoints `/chat` SSE et `/reset`). Le streaming réel vient de llama.cpp (`stream=True`).

**Tech Stack:** Python 3.12+, `uv`, `pytest`, `openai` (SDK Stainless), `flask`, HTMX (vendored, offline).

**Spec de référence:** [docs/superpowers/specs/2026-05-31-loom-chat-design.md](../specs/2026-05-31-loom-chat-design.md)

> Projet **hors git** : sauter toutes les étapes `git`. Toolchain `uv`. Aucune commande ne doit
> appeler un vrai serveur/modèle dans les tests (tout est mocké).

```python
# loom/conversation.py
"""Mémoire de conversation : historique des messages + persistance JSON."""
from __future__ import annotations

import json
from dataclasses import dataclass, field
from pathlib import Path


@dataclass
class Conversation:
    system_prompt: str
    messages: list[dict] = field(default_factory=list)

    def add(self, role: str, content: str) -> None:
        self.messages.append({"role": role, "content": content})

    def reset(self) -> None:
        self.messages = []

    def to_messages(self) -> list[dict]:
        return list(self.messages)

    def save(self, path: str | Path) -> None:
        path = Path(path)
        path.parent.mkdir(parents=True, exist_ok=True)
        data = {"system_prompt": self.system_prompt, "messages": self.messages}
        path.write_text(json.dumps(data, ensure_ascii=False, indent=2), encoding="utf-8")

    @classmethod
    def load(cls, path: str | Path, default_system_prompt: str) -> "Conversation":
        path = Path(path)
        if not path.exists():
            return cls(system_prompt=default_system_prompt)
        try:
            data = json.loads(path.read_text(encoding="utf-8"))
            return cls(
                system_prompt=data.get("system_prompt", default_system_prompt),
                messages=list(data.get("messages", [])),
            )
        except (json.JSONDecodeError, OSError):
            return cls(system_prompt=default_system_prompt)
```

```python
# loom/client.py
"""Client modèle : parle à l'endpoint OpenAI-compatible de Loom via le SDK openai."""
from __future__ import annotations

from collections.abc import Iterator

from openai import OpenAI


def _iter_deltas(stream) -> Iterator[str]:
    """Extrait les morceaux de texte non vides d'un flux de chunks openai."""
    for chunk in stream:
        delta = chunk.choices[0].delta.content
        if delta:
            yield delta


class LoomClient:
    def __init__(
        self,
        base_url: str,
        api_key: str = "loom-local",
        model: str = "local",
    ) -> None:
        self.base_url = base_url
        self.model = model
        self._client = OpenAI(base_url=base_url, api_key=api_key)

    def stream_chat(self, messages: list[dict], system_prompt: str) -> Iterator[str]:
        """Yield les tokens de la réponse, system prompt injecté en tête."""
        full = [{"role": "system", "content": system_prompt}, *messages]
        stream = self._client.chat.completions.create(
            model=self.model, messages=full, stream=True
        )
        yield from _iter_deltas(stream)
```

```python
"""Application Flask de Loom Chat : page + endpoints chat/reset."""
from __future__ import annotations

import json
from pathlib import Path

from flask import Flask, Response, render_template, request


def create_app(conversation, client, history_path: str | Path) -> Flask:
    app = Flask(__name__)
    history_path = str(history_path)

    @app.get("/")
    def index() -> str:
        return render_template("index.html", messages=conversation.messages)

    @app.post("/reset")
    def reset() -> str:
        conversation.reset()
        conversation.save(history_path)
        return render_template("index.html", messages=conversation.messages)

    return app
```

```python
    @app.post("/chat")
    def chat():
        message = (request.form.get("message") or "").strip()
        if not message or len(message) > 5000:
            return Response("message invalide", status=400)

        conversation.add("user", message)
        conversation.save(history_path)

        def generate():
            full = ""
            try:
                for token in client.stream_chat(
                    conversation.to_messages(), conversation.system_prompt
                ):
                    full += token
                    yield f"data: {json.dumps({'type': 'text', 'text': token})}\n\n"

                yield f"data: {json.dumps({'type': 'done'})}\n\n"
            except Exception as exc:  # noqa: BLE001 - on remonte l'erreur au client SSE
                yield f"data: {json.dumps({'type': 'error', 'message': str(exc)})}\n\n"

        return Response(generate(), mimetype="text/event-stream")
```

```html
      e.preventDefault();
      const text = input.value.trim();
      if (!text) return false;

        if (done) break;

            window.scrollTo(0, document.body.scrollHeight);

            assistant.textContent = 'Erreur : ' + evt.message +
              ' — Loom est-il lancé ? (uv run loom/serve.py)';
          }
        }
      }
      return false;
    }
```

```python
    app = create_app(conversation, client, cfg.chat.history_path)
    print(f"[loom-chat] http://127.0.0.1:{cfg.chat.web_port}  (modèle: {base_url})")
    app.run(host="127.0.0.1", port=cfg.chat.web_port)


if __name__ == "__main__":
```
